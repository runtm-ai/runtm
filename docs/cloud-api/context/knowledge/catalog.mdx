---
title: "Knowledge Catalog"
api: "GET /api/knowledge/catalog"
description: "List every available knowledge provider and its capabilities."
---

Returns the full catalog of knowledge providers with their auth methods, tools, and display metadata.
The catalog is resolved from code at import time - no database queries are involved.

**Required scope:** `integrations:read`

<ResponseField name="providers" type="CatalogProviderResource[]" required>
  List of available knowledge providers.

  <Expandable title="CatalogProviderResource">
    <ResponseField name="id" type="string" required>Provider identifier (e.g. `"bigquery"`).</ResponseField>
    <ResponseField name="display_name" type="string" required>Human-readable name.</ResponseField>
    <ResponseField name="tagline" type="string" required>Short tagline for the provider picker.</ResponseField>
    <ResponseField name="description" type="string" required>Full description.</ResponseField>
    <ResponseField name="category" type="string" required>Grouping category.</ResponseField>
    <ResponseField name="icon" type="string" required>Icon identifier.</ResponseField>
    <ResponseField name="accent_color" type="string" required>CSS color value.</ResponseField>
    <ResponseField name="docs_url" type="string">Link to provider documentation.</ResponseField>
    <ResponseField name="auth_methods" type="CatalogAuthMethodResource[]" required>
      Supported authentication methods.

      <Expandable title="CatalogAuthMethodResource">
        <ResponseField name="id" type="string" required>Method identifier.</ResponseField>
        <ResponseField name="display_name" type="string" required>Human-readable name.</ResponseField>
        <ResponseField name="description" type="string" required>Short description.</ResponseField>
        <ResponseField name="auth_kind" type="string" required>`"oauth"` or `"static"`.</ResponseField>
        <ResponseField name="oauth_scopes" type="string[]">OAuth scopes requested (OAuth methods only).</ResponseField>
        <ResponseField name="credential_schema" type="object">JSON Schema for static credentials.</ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="tools" type="CatalogToolResource[]" required>
      Tools the provider exposes.

      <Expandable title="CatalogToolResource">
        <ResponseField name="id" type="string" required>Tool identifier.</ResponseField>
        <ResponseField name="name" type="string" required>Human-readable tool name.</ResponseField>
        <ResponseField name="description" type="string" required>What the tool does.</ResponseField>
        <ResponseField name="kind" type="string" required>`"read"` or `"write"`.</ResponseField>
        <ResponseField name="default_permission" type="string" required>`"auto"`, `"allow"`, or `"ask"`.</ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
