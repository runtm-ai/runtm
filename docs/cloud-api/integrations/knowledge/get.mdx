---
title: "Get Knowledge Integration"
api: "GET /knowledge/integrations/{integration_id}"
description: "Get a single knowledge integration by id"
---

Returns a single knowledge integration. The canonical resource shape returned here is the same shape every other endpoint in this section returns (List, Create, Patch, Rotate, Test).

Credentials are **never** returned. Callers can only learn whether one is set via the `has_credentials` field.

<Info>Required scope: `integrations:read`</Info>

<Note>
This endpoint is **organization-only**. The request must include `X-Organization-Id` (or use an org-scoped API key).
</Note>

## Path Parameters

<ParamField path="integration_id" type="string" required>
  Knowledge integration id (e.g. `kint_aaa`).
</ParamField>

## Headers

<ParamField header="X-Organization-Id" type="string" required>
  The organization context.
</ParamField>

## Response

<ResponseField name="integration" type="object">
  <Expandable title="KnowledgeIntegrationResource">
    <ResponseField name="id" type="string">
      Stable integration id.
    </ResponseField>
    <ResponseField name="organization_id" type="string">
      Organization that owns the connection.
    </ResponseField>
    <ResponseField name="user_id" type="string">
      `null` for org-shared connections; non-null for personal overrides.
    </ResponseField>
    <ResponseField name="provider" type="string">
      Provider id (e.g. `bigquery`).
    </ResponseField>
    <ResponseField name="auth_method" type="string">
      Auth method id from the catalog.
    </ResponseField>
    <ResponseField name="display_name" type="string">
      Optional human-readable label.
    </ResponseField>
    <ResponseField name="status" type="string">
      `active`, `revoked`, `error`, or `pending`.
    </ResponseField>
    <ResponseField name="has_credentials" type="boolean">
      True if a credential is currently set. The actual credential value is never returned.
    </ResponseField>
    <ResponseField name="has_oauth_refresh" type="boolean">
      True if an OAuth refresh token is stored.
    </ResponseField>
    <ResponseField name="oauth_access_expires_at" type="string">
      ISO 8601 expiry of the cached OAuth access token (if applicable).
    </ResponseField>
    <ResponseField name="provider_metadata" type="object">
      Free-form non-secret metadata.
    </ResponseField>
    <ResponseField name="default_mode" type="string">
      Default per-tool permission (`auto`, `allow`, `ask`).
    </ResponseField>
    <ResponseField name="tool_permissions" type="object">
      Per-tool permission overrides keyed by tool id.
    </ResponseField>
    <ResponseField name="connected_by_user_id" type="string">
      User who originally connected the integration.
    </ResponseField>
    <ResponseField name="last_used_at" type="string">
      ISO 8601 timestamp of the integration's last use, or `null`.
    </ResponseField>
    <ResponseField name="created_at" type="string">
      ISO 8601 creation timestamp.
    </ResponseField>
    <ResponseField name="updated_at" type="string">
      ISO 8601 last-update timestamp, or `null`.
    </ResponseField>
  </Expandable>
</ResponseField>
